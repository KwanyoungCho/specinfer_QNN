find_package(OpenCL REQUIRED)
find_package(Python3 REQUIRED)

set(TARGET speculative-eagle-opencl-tuner)
set(OPENCL_KERNEL_DIR "${CMAKE_SOURCE_DIR}/ggml/src/ggml-opencl/kernels")
set(EMBED_KERNEL_SCRIPT "${OPENCL_KERNEL_DIR}/embed_kernel.py")
set(GENERATED_KERNEL_DIR "${CMAKE_CURRENT_BINARY_DIR}/autogenerated")

file(MAKE_DIRECTORY "${GENERATED_KERNEL_DIR}")

foreach(KERNEL_NAME argsort set_rows)
    add_custom_command(
        OUTPUT "${GENERATED_KERNEL_DIR}/${KERNEL_NAME}.cl.h"
        COMMAND ${Python3_EXECUTABLE} "${EMBED_KERNEL_SCRIPT}"
                "${OPENCL_KERNEL_DIR}/${KERNEL_NAME}.cl"
                "${GENERATED_KERNEL_DIR}/${KERNEL_NAME}.cl.h"
        DEPENDS "${OPENCL_KERNEL_DIR}/${KERNEL_NAME}.cl" "${EMBED_KERNEL_SCRIPT}"
        COMMENT "Generate embedded OpenCL kernel header ${KERNEL_NAME}.cl.h"
    )
endforeach()

add_executable(${TARGET} main.cpp)
target_sources(${TARGET} PRIVATE
    "${GENERATED_KERNEL_DIR}/argsort.cl.h"
    "${GENERATED_KERNEL_DIR}/set_rows.cl.h")
target_link_libraries(${TARGET} PRIVATE ${OpenCL_LIBRARIES})
target_include_directories(${TARGET} PRIVATE
    ${OpenCL_INCLUDE_DIRS}
    "${GENERATED_KERNEL_DIR}")
target_compile_features(${TARGET} PRIVATE cxx_std_17)
target_compile_definitions(${TARGET} PRIVATE
    SPECINFER_QNN_OPENCL_EMBED_KERNELS
    SPECINFER_QNN_OPENCL_KERNEL_DIR="${CMAKE_SOURCE_DIR}/ggml/src/ggml-opencl/kernels")

if(LLAMA_TOOLS_INSTALL)
    install(TARGETS ${TARGET} RUNTIME)
endif()
